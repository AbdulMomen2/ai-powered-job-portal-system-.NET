    <script src="https://unpkg.com/lucide@latest"></script>
    <script>
        lucide.createIcons();

        // ── Review Slider ──────────────────────────────────────────────
        const track = document.getElementById('reviewTrack');
        const dotsContainer = document.getElementById('reviewDots');
        const slides = track ? track.querySelectorAll('.review-slide') : [];
        const CARD_W = 320 + 20; // card width + gap
        const VISIBLE = Math.max(1, Math.floor(window.innerWidth / CARD_W));
        const MAX_IDX = slides.length > VISIBLE ? slides.length - VISIBLE : 0;
        let reviewIdx = 0;
        let reviewTimer = null;

        function buildDots() {
            if (!dotsContainer) return;
            dotsContainer.innerHTML = '';
            for (let i = 0; i <= MAX_IDX; i++) {
                const d = document.createElement('button');
                d.className = `w-2 h-2 rounded-full transition-all duration-300 ${i === 0 ? 'bg-primary w-5' : 'bg-gray-300'}`;
                d.onclick = () => goToReview(i);
                dotsContainer.appendChild(d);
            }
        }

        function updateDots() {
            if (!dotsContainer) return;
            dotsContainer.querySelectorAll('button').forEach((d, i) => {
                d.className = `h-2 rounded-full transition-all duration-300 ${i === reviewIdx ? 'bg-primary w-5' : 'bg-gray-300 w-2'}`;
            });
        }

        function goToReview(idx) {
            reviewIdx = Math.max(0, Math.min(idx, MAX_IDX));
            if (track) track.style.transform = `translateX(-${reviewIdx * CARD_W}px)`;
            const prev = document.getElementById('reviewPrev');
            const next = document.getElementById('reviewNext');
            if (prev) prev.disabled = reviewIdx === 0;
            if (next) next.disabled = reviewIdx === MAX_IDX;
            updateDots();
        }

        function shiftReviews(dir) {
            goToReview(reviewIdx + dir);
            restartAutoSlide();
        }

        function restartAutoSlide() {
            clearInterval(reviewTimer);
            reviewTimer = setInterval(() => {
                goToReview(reviewIdx < MAX_IDX ? reviewIdx + 1 : 0);
            }, 4000);
        }

        buildDots();
        restartAutoSlide();

        // ── Slide-over ─────────────────────────────────────────────────
        const slideOver = document.getElementById('jobSlideOver');
        const slideOverPanel = document.getElementById('slideOverPanel');
        const slideOverBackdrop = document.getElementById('slideOverBackdrop');

        function openJobDetails(el) {
            if (el) {
                document.getElementById('soTitle').textContent = el.dataset.title || '';
                document.getElementById('soLocation').innerHTML = `<i data-lucide="map-pin" class="w-4 h-4 text-gray-400 inline mr-1"></i>${el.dataset.location || ''}`;
                document.getElementById('soSalary').innerHTML = `<i data-lucide="dollar-sign" class="w-4 h-4 text-gray-400 inline mr-1"></i>${el.dataset.salary || ''}`;
                document.getElementById('soDesc').innerHTML = el.dataset.desc || '<p>No description provided.</p>';
                lucide.createIcons();
            }
            slideOver.classList.remove('hidden');
            setTimeout(() => {
                slideOverBackdrop.classList.remove('opacity-0');
                slideOverBackdrop.classList.add('opacity-100');
                slideOverPanel.classList.remove('slide-over-closed');
                slideOverPanel.classList.add('slide-over-open');
            }, 10);
            document.body.style.overflow = 'hidden';
        }

        function closeJobDetails() {
            slideOverBackdrop.classList.remove('opacity-100');
            slideOverBackdrop.classList.add('opacity-0');
            slideOverPanel.classList.remove('slide-over-open');
            slideOverPanel.classList.add('slide-over-closed');
            setTimeout(() => {
                slideOver.classList.add('hidden');
                document.body.style.overflow = '';
            }, 300);
        }

        // ── Auth Modal ─────────────────────────────────────────────────
        const authModal = document.getElementById('authModal');
        let authDismissed = sessionStorage.getItem('authDismissed') === '1';
        let countdown = 60;
        let countdownTimer = null;

        function openAuthModal(intent) {
            if (!authModal) return;
            const title = document.getElementById('authModalTitle');
            const desc = document.getElementById('authModalDesc');
            const loginBtn = document.getElementById('authLoginBtn');
            const regBtn = document.getElementById('authRegisterBtn');

            if (intent === 'employer') {
                if (title) title.textContent = 'Sign in as Employer';
                if (desc) desc.textContent = 'Log in or create an employer account to post jobs and find top talent.';
                if (loginBtn) loginBtn.href = '/Account/Login?returnUrl=/Employer/CreateJob&role=Employer';
                if (regBtn) regBtn.href = '/Account/Register?role=Employer';
            } else {
                if (title) title.textContent = 'Sign in to continue';
                if (desc) desc.textContent = 'Create a free account or log in to apply for jobs, track applications, and more.';
                if (loginBtn) loginBtn.href = '/Account/Login';
                if (regBtn) regBtn.href = '/Account/Register';
            }
            authModal.classList.remove('hidden');
            authDismissed = false;
        }

        function dismissAuthModal() {
            if (!authModal) return;
            authModal.classList.add('hidden');
            authDismissed = true;
            sessionStorage.setItem('authDismissed', '1');
            clearInterval(countdownTimer);
        }

        // 60-second auto-trigger for guests
        if (authModal && !authDismissed) {
            let secs = 60;
            const numEl = document.getElementById('countdownNum');
            const ringWrapper = document.getElementById('countdownRingWrapper');
            countdownTimer = setInterval(() => {
                secs--;
                if (numEl) numEl.textContent = secs;
                if (secs <= 0) {
                    clearInterval(countdownTimer);
                    if (!authDismissed) {
                        if (ringWrapper) ringWrapper.classList.remove('hidden');
                        openAuthModal('candidate');
                    }
                }
            }, 1000);
        }

        // Escape key closes slide-over and modal
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                closeJobDetails();
                dismissAuthModal();
            }
        });
    </script>
