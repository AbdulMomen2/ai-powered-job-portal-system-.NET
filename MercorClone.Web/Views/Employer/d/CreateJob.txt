@section Scripts {
    @{await Html.RenderPartialAsync("_ValidationScriptsPartial");}

    <script>
        // 1. Initialize Rich Text Editor
        tinymce.init({
            selector: '#editor',
            height: 300,
            menubar: false,
            plugins: 'lists link',
            toolbar: 'bold italic underline | bullist numlist | link',
            setup: function (editor) {
                editor.on('change', function () {
                    tinymce.triggerSave(); // Sync to textarea for submission
                });
            }
        });

        // 2. Wizard Navigation Logic
        function nextStep(current, next) {
            // Basic Frontend Validation (Check required fields)
            let currentDiv = document.getElementById('step' + current);
            let inputs = currentDiv.querySelectorAll('input[required]');
            let isValid = true;
            
            inputs.forEach(input => {
                if(!input.value) {
                    input.classList.add('is-invalid');
                    isValid = false;
                } else {
                    input.classList.remove('is-invalid');
                }
            });

            if(!isValid) return; // Stop if validation fails

            // Hide current, show next
            document.getElementById('step' + current).classList.remove('active');
            document.getElementById('step' + next).classList.add('active');
            window.scrollTo(0, 0); // Scroll to top
        }

        function prevStep(current, prev) {
            document.getElementById('step' + current).classList.remove('active');
            document.getElementById('step' + prev).classList.add('active');
            window.scrollTo(0, 0);
        }

        // 3. Populate Review Screen dynamically before showing it
        function prepareReview() {
            // Sync TinyMCE data to textarea
            tinymce.triggerSave();

            // Populate summary fields
            document.getElementById('revTitle').innerText = document.getElementById('Title').value;
            
            let locType = document.getElementById('LocationType').value;
            let address = document.getElementById('Location').value;
            document.getElementById('revLocation').innerText = locType === 'Remote' ? 'Remote' : address + ' (' + locType + ')';

            document.getElementById('revType').innerText = document.querySelector('input[name="ContractType"]:checked').value;
            
            let min = document.getElementById('MinSalary').value;
            let max = document.getElementById('MaxSalary').value;
            document.getElementById('revPay').innerText = min && max ? `$${min} - $${max}` : 'Not Specified';
            
            document.getElementById('revHires').innerText = document.getElementById('HiresTarget').value;

            // Move to Step 5
            nextStep(4, 5);
        }

        // 4. Dynamic Location Input (Hide street address if Remote)
        document.getElementById('LocationType').addEventListener('change', function() {
            let locField = document.getElementById('locationField');
            if(this.value === 'Remote') {
                locField.style.display = 'none';
                document.getElementById('Location').removeAttribute('required');
            } else {
                locField.style.display = 'block';
                document.getElementById('Location').setAttribute('required', 'required');
            }
        });
    </script>
}
